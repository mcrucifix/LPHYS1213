% print automatically all section names in format 0*.tex and all appendices in format A_*

\begin{luacode*}
  files = {}
  for file in lfs.dir(".") do
  if file:match("^0%d_.*%.tex$") then
      table.insert(files, (file:gsub("%.tex$", "")))
    end
  end

  table.sort(files)

  for _, f in ipairs(files) do
    tex.print(-1, "\\include{" .. f .. "}")
  end

\end{luacode*}
